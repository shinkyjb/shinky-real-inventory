CREATE TABLE IF NOT EXISTS items (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  name text NOT NULL,
  game text NOT NULL,
  type text NOT NULL,
  price numeric NOT NULL,
  status text NOT NULL,
  spec text,
  created_at timestamptz DEFAULT now()
);

CREATE TABLE IF NOT EXISTS item_images (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  item_id UUID REFERENCES items(id) ON DELETE CASCADE,
  path text NOT NULL,
  created_at timestamptz DEFAULT now()
);

ALTER TABLE items ENABLE ROW LEVEL SECURITY;
ALTER TABLE item_images ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Anyone can read items" ON items FOR SELECT USING (true);
CREATE POLICY "Anyone can read images" ON item_images FOR SELECT USING (true);
CREATE POLICY "Authenticated users can insert items" ON items FOR INSERT WITH CHECK (auth.role() = 'authenticated');
CREATE POLICY "Authenticated users can insert images" ON item_images FOR INSERT WITH CHECK (auth.role() = 'authenticated');

CREATE INDEX IF NOT EXISTS items_game_idx ON items(game);
CREATE INDEX IF NOT EXISTS items_status_idx ON items(status);
CREATE INDEX IF NOT EXISTS item_images_item_id_idx ON item_images(item_id);
